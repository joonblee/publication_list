\begin{pubentry}{P1}{Searching for New Physics Inside Jets with the Herwig 7 Generalised Parton Shower}
\pubauthors{Taehee Kim, \pubme\corr, M. R. Masouminia, Michael H. Seymour and Un-ki Yang.}
\pubdetails{Accepted for publication in \textit{JHEP} (2026). \arxivlink{2604.00617}.}
\contribution{I \keycontribution{conceived and led the study from proposal through peer review}. I designed the benchmarks and analysis strategy, performed theoretical and numerical studies, and led manuscript preparation and referee responses. Using the BSM shower framework I developed~\pubref{P2}, I \keycontribution{characterised non-isolated dimuon signatures inside jets}, identified discriminating observables and evaluated prospective sensitivity. The study establishes the phenomenological basis for the CMS dimuon-in-jet search\ifincludeinprogress~\pubref{R1}\fi.}
\end{pubentry}
% Keep the emphasis on scientific and manuscript leadership. Routine sample

% P2: The original CMS search motivation was proposed by Un-ki Yang;
% the generalised BSM-shower project and MCnet proposal were J.-B. Lee's.
\begin{pubentry}{P2}{Generalized angular-order parton showers in Herwig 7}
\pubauthors{\pubme\corr, M. R. Masouminia, Michael H. Seymour and Un-ki Yang.}
\pubdetails{\textit{JHEP} \textbf{08} (2024) 064. \arxivlink{2312.13125}; \doilink{10.1007/JHEP08(2024)064}.}
\contribution{Prompted by Un-ki Yang's proposal for a dimuon-in-$b$-jet search\ifincludeinprogress~\pubref{R1}\fi, I identified BSM shower radiation as a possible signal-production mechanism, \keycontribution{proposed the generalised BSM-shower project} and secured an MCnet studentship. Under Michael H. Seymour's supervision, I \keycontribution{derived the splitting functions and implemented the shower algorithms}, including automatic UFO-based model support, and led validation and manuscript preparation. The framework enables the signal modelling used in~\pubref{P1} and the CMS search. A \href{https://github.com/joonblee/herwig74alpha}{public development version} is available ahead of the official Herwig~7.4 release\ifincludeinprogress~\pubref{R3}\fi, providing a basis for future-collider studies and further work on matching and merging.}
\end{pubentry}

\begin{pubentry}{C1}{Implementation of the CMS-HIG-18-011 analysis in the MadAnalysis 5 framework (exotic Higgs decays via two pseudoscalars with two muons and two $b$-jets; 35.9~fb$^{-1}$)}

\pubdetails{\textit{Mod. Phys. Lett. A} \textbf{36} (2021) 2141003 (peer-reviewed conference proceeding).\\

\end{pubentry}

\section*{ENGINEERING --- PUBLISHED}

\begin{pubentry}{P3}{Analytical modelling of frequency splitting and modal degeneracy restoration in imperfect wine-glass shell resonators}
\pubauthors{\pubme.}
\pubdetails{\textit{Int. J. Eng. Sci.} \textbf{227} (2026) 104593. \doilink{10.1016/j.ijengsci.2026.104593}.}
\contribution{Motivated by possible new-physics probes beyond the LHC, I applied my background in aerospace structures and vibration to high-quality-factor shell resonators. Structural imperfections split otherwise equal frequencies within pairs of vibration modes, limiting sensor performance. As sole author, I derived the shell-vibration model and \keycontribution{a geometrical balancing method that restores equal frequencies within mode pairs} (modal degeneracy), and validated the results with numerical calculations and COMSOL simulations. By balancing added and removed mass, the method offers a route to reduce reliance on auxiliary tuning hardware. My longer-term goal is to develop this approach for BSM sensing.}
\end{pubentry}
