{\fontsize{10.5pt}{12.5pt}\selectfont
My research combines event-generator development, collider phenomenology and CMS data analysis to investigate unconventional signatures of new physics. The dimuon-in-jet programme connects studies of muons and $b$~jets, BSM shower modelling and signature characterisation to an experimental search, supported by broader generator integration and modelling work. Complementary studies of shell resonators explore mechanical foundations for possible non-collider probes of new physics.\par}

\section*{PARTICLE PHYSICS --- PUBLISHED OR ACCEPTED}

% P1: The author proposed and led the study, including its scientific and
% manuscript strategy. Leadership does not imply sole execution by one author.
\begin{pubentry}{P1}{Searching for New Physics Inside Jets with the Herwig 7 Generalised Parton Shower}
\pubauthors{Taehee Kim, \pubme\corr, M. R. Masouminia, Michael H. Seymour and Un-ki Yang.}
\pubdetails{Accepted for publication in \textit{JHEP} (2026). \arxivlink{2604.00617}.}
\contribution{I conceived and led the study from the research proposal through manuscript preparation and referee responses. I designed the benchmarks and analysis strategy, performed theoretical and numerical studies, and generated signal and background samples with collaborator support. Using the BSM shower framework I developed, I characterised non-isolated dimuon signatures inside jets, identified discriminating observables and evaluated prospective sensitivity. This work connects generator development to experimentally testable search strategies for the CMS dimuon-in-jet programme.}
\end{pubentry}

% P2: The original CMS search motivation was proposed by Un-ki Yang;
% the generalised BSM-shower project and MCnet proposal were J.-B. Lee's.
\begin{pubentry}{P2}{Generalized angular-order parton showers in Herwig 7}
\pubauthors{\pubme\corr, M. R. Masouminia, Michael H. Seymour and Un-ki Yang.}
\pubdetails{\textit{JHEP} \textbf{08} (2024) 064. \arxivlink{2312.13125}; \doilink{10.1007/JHEP08(2024)064}.}
\contribution{I identified the simulation requirements of the dimuon-in-jet search, proposed the generalised BSM-shower project and secured an MCnet studentship to pursue it. I derived BSM splitting functions, implemented the C++ shower algorithms and UFO-based model support, and led physics validation and numerical studies. I prepared the first manuscript and subsequent revisions under Michael H. Seymour's scientific supervision. The framework supplies the radiation-based signal modelling needed for the phenomenological and experimental programme.}
\end{pubentry}




\begin{pubentry}{C1}{Implementation of the CMS-HIG-18-011 analysis in the MadAnalysis 5 framework (exotic Higgs decays via two pseudoscalars with two muons and two $b$-jets; 35.9~fb$^{-1}$)}
\pubauthors{\pubme\corr\ and Jehyun Lee.}
\pubdetails{\textit{Mod. Phys. Lett. A} \textbf{36} (2021) 2141003 (peer-reviewed conference proceeding).\\
\doilink{10.1142/S0217732321410030}.}
\contribution{I reimplemented the published CMS analysis in MadAnalysis~5, including object selection, detector-response modelling, sample generation, cut-flow and efficiency validation, and statistical interpretation. I produced the results and initial manuscript with guidance from the MadAnalysis~5 team. This work established my practical understanding of CMS muons, $b$-jet identification and reconstruction effects, providing a foundation for developing the subsequent non-isolated dimuon search.}
\end{pubentry}

\section*{COMPLEMENTARY RESEARCH --- PUBLISHED}

\begin{pubentry}{P3}{Analytical modelling of frequency splitting and modal degeneracy restoration in imperfect wine-glass shell resonators}
\pubauthors{\pubme.}
\pubdetails{\textit{Int. J. Eng. Sci.} \textbf{227} (2026) 104593. \doilink{10.1016/j.ijengsci.2026.104593}.}
\contribution{As sole author, I formulated the research problem, derived the shell-vibration model and an imperfection-balancing method to restore modal degeneracy, and validated the analytical results with numerical calculations and COMSOL simulations. I wrote the manuscript and referee responses. Motivated by my interest in unconventional probes beyond the LHC, this study develops a mechanical-modelling foundation for resonator-based sensing; applications to BSM detection remain prospective.}
\end{pubentry}

\begin{pubentry}{R1}{Search for anomalous dimuon resonances within b jets in proton-proton collisions at $\sqrt{s}=13$ and 13.6~TeV}
\pubauthors{CMS Collaboration --- internal analysis; analysis contact: \pubme.}
\pubdetails{\textbf{CMS-NPS-26-009}. \npsstatus}
\contribution{I lead the experimental search and serve as analysis contact. I designed the signal and control regions, optimised muon and $b$-jet selections, and developed the analysis code and signal-simulation workflow. I performed background estimation, systematic-uncertainty evaluation and statistical inference, including Combine datacards and limit calculations. I wrote the principal Analysis Note content and coordinate the ongoing CMS internal-review responses. This analysis brings together my reconstruction studies, BSM-shower development and phenomenology to investigate non-isolated dimuon resonances in CMS collision data.}
\end{pubentry}

\begin{pubentry}{R2}{Modelling physics processes with MadGraph5\_aMCatNLO and parton shower Monte Carlos in the CMS experiment}
\pubauthors{CMS Collaboration --- internal manuscript; analysis contact: \pubme.}
\pubdetails{\textbf{CMS-GEN-25-002}. \genstatus}
