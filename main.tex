\pubauthors{\pubme\corr, M. R. Masouminia, Michael H. Seymour and Un-ki Yang.}
\pubdetails{\textit{JHEP} \textbf{08} (2024) 064. \arxivlink{2312.13125}; \doilink{10.1007/JHEP08(2024)064}.}

\contribution{Prompted by Un-ki Yang's proposal for a dimuon-in-$b$-jet search\ifincludeinprogress~\pubref{R1}\fi, I identified BSM shower radiation as a possible signal-production mechanism, \keycontribution{proposed the generalised BSM-shower project} and received an MCnet studentship. It enables the signal modelling used in~\pubref{P1} and the CMS search. \section*{ENGINEERING --- PUBLISHED}

\begin{pubentry}{P3}{Analytical modelling of frequency splitting and modal degeneracy restoration in imperfect wine-glass shell resonators}
\pubauthors{\pubme.}
\pubdetails{\textit{Int. J. Eng. Sci.} \textbf{227} (2026) 104593. \doilink{10.1016/j.ijengsci.2026.104593}.}
\contributionitems{\checkitem{Research design}\checkitem{Analytical model}\checkitem{Geometrical balancing}\checkitem{Numerical calculations}\checkitem{COMSOL validation}\checkitem{Manuscript \& referee responses}}
\contribution{As sole author, I applied my background in aerospace structures and vibration to high-$Q$ shell resonators for possible new-physics probes beyond the LHC. Imperfections split otherwise equal vibration-mode frequencies; compensation can require costly tuning equipment or additional control hardware. I \keycontribution{developed a geometrical method to restore equal frequencies within mode pairs} (modal degeneracy). Balancing added and removed mass offers a route to reduce these cost and hardware barriers, supporting my longer-term goal of BSM sensing.}
\end{pubentry}
